server {
    listen 80;
    listen [::]:80;

    root /var/www/html;  # Make sure this is the path to your website's code

    index index.html index.htm index.nginx-debian.html;  # Ensure your index file is listed here

    server_name devcom.web.tr www.devcom.web.tr;  # Replace _ with your domain name

    location / {
        try_files $uri $uri/ =404;
    }

    # Optional: Redirect www to non-www
    if ($host = 'www.devcom.web.tr') {
        return 301 http://devcom.web.tr$request_uri;
    }
}
